cmake_minimum_required(VERSION 3.22.1)
project(englishdub LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/whisper)
add_library(englishdub SHARED whisper_jni.cpp)
target_link_libraries(englishdub PRIVATE whisper android log)
target_link_options(englishdub PRIVATE "-Wl,-z,max-page-size=16384")
