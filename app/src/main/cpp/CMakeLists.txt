cmake_minimum_required(VERSION 3.22.1)
project(englishdub LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    # Neural kernels at -O0 distort QA/profiling and can take minutes for a short clip.
    # Keep debug symbols and assertions while using the optimized kernel implementation.
    add_compile_options(-O2)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "x86")
    # Android's legacy x86 emulator exposes SSE3 but not AVX/FMA instructions.
    foreach(feature AVX AVX2 FMA F16C AVX_VNNI AVX512 SSE42)
        set(GGML_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
endif()
add_subdirectory(vendor/whisper)
add_library(englishdub SHARED whisper_jni.cpp)
target_link_libraries(englishdub PRIVATE whisper android log)
target_link_options(englishdub PRIVATE "-Wl,-z,max-page-size=16384")
